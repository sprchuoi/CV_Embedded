% Section formatting for main content
\titleformat{\section}
{\normalfont\large\bfseries\color{textdark}}
{}{0em}{}

% Sidebar section command
\newcommand{\sidebarsection}[1]{%
    \vspace{0.5em}
    {\large\bfseries\textcolor{textlight}{#1}}\\[0.3em]
    \textcolor{accentcolor}{\rule{0.9\linewidth}{1pt}}
    \vspace{0.3em}
}

% Skill item with icon. Starts with \par so a category always begins on a new
% line -- a newline in the LaTeX source is only a space, so a preceding tag row
% would otherwise run into this heading on the same line.
\newcommand{\skillcategory}[2]{%
    \par\vspace{0.2em}%
    \textcolor{accentcolor}{\small\faIcon{#1}} \textbf{\textcolor{textlight}{#2}}\\[0.2em]
}

% Skill tags. Solid accent fill with white text for contrast against the dark
% sidebar; the previous 20%-tinted fill made the light label nearly invisible.
\newcommand{\skilltag}[1]{%
    \tikz[baseline=(char.base)]{
        \node[fill=accentcolor, text=white, rounded corners=2pt, inner sep=2pt, font=\scriptsize] (char) {#1};
    }%
}

% Contact item
\newcommand{\contactitem}[2]{%
    \textcolor{accentcolor}{\faIcon{#1}} \textcolor{textlight}{\small #2}\\[0.3em]
}

% Experience entry
\newcommand{\experienceentry}[4]{%
    \textbf{#1}\\
    {\small\textcolor{subtextcolor}{#2 | #3}}\\[0.2em]
    #4
    \vspace{0.4em}
}

% Education entry
\newcommand{\educationentry}[4]{%
    \textbf{#1}\\
    {\small\textcolor{subtextcolor}{#2 | #3}}\\
    {\small #4}
    \vspace{0.3em}
}

% Nested project under an experience entry
\newcommand{\projectentry}[2]{%
    \par\vspace{0.2em}
    {\small\bfseries\color{accentcolor}#1}\\[0.1em]
    #2
    \vspace{0.2em}
}

% A "<b>Label:</b> value" field inside an entry
\newcommand{\entryfield}[2]{%
    {\small\textbf{#1:}\ifx\relax#2\relax\else\ #2\fi}\\[0.15em]
}

% Prose-valued skill category (a sentence rather than a tag list)
\newcommand{\skillprose}[1]{%
    \par\textcolor{textlight}{\scriptsize\itshape #1}\par\vspace{0.1em}
}

% Award line for the LEFT sidebar (light text on the dark background).
\newcommand{\awarditem}[2]{%
    \textcolor{accentcolor}{\faIcon{#1}} \textcolor{textlight}{\small #2}\\[0.2em]
}

% Achievement line for the RIGHT column, which is on white paper. Needs dark
% text -- reusing \awarditem here rendered light-on-white and was invisible.
\newcommand{\achievementitem}[2]{%
    \par\textcolor{accentcolor}{\faIcon{#1}} \textcolor{textdark}{\small #2}\par\vspace{0.2em}
}
\newcommand{\awarddetail}[1]{%
    \hspace{1em}\textcolor{textlight}{\scriptsize #1}\\[0.15em]
}

% Compact itemize
\setlist[itemize]{
    leftmargin=1em,
    itemsep=0pt,
    parsep=0pt,
    topsep=0.2em,
    partopsep=0pt,
    label={\textcolor{accentcolor}{\textbullet}}
}

% Full-bleed sidebar band, painted on EVERY page.
% paracol's own \backgroundcolor only covers each column's own height, so on a
% short final page the dark band stopped partway down and left white below it.
% Painting the band per-page decouples it from the column content height.
\AddToShipoutPictureBG{%
    \begin{tikzpicture}[remember picture, overlay]
        \fill[sidebarcolor]
            (current page.north west)
            rectangle ([xshift=0.32\paperwidth]current page.south west);
    \end{tikzpicture}%
}
